# SMS via Brevo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add SMS delivery via Brevo Transactional SMS so clients receive checklist notifications by text message alongside existing email notifications.

**Architecture:** Create `lib/notifications/channels/sms.ts` with a `sendSms()` function mirroring the existing `email.ts` pattern (same retry logic, same env access). Extend the send-notification worker to handle `channel === 'sms'`. Add seed SQL for SMS message templates. No DB schema changes needed — the schema already supports `channel = 'sms'` everywhere.

**Tech Stack:** Brevo Transactional SMS API (`/v3/transactionalSMS/sms`), Vitest, TypeScript, Zod, Next.js App Router API routes.

- [ ] **Step 3: Create `lib/notifications/channels/sms.ts`**

```ts
import { getEnv } from '@/lib/env'

const BREVO_SMS_API = 'https://api.brevo.com/v3/transactionalSMS/sms'

interface SendSmsParams {
  to: string
  message: string
}

export async function sendSms({ to, message }: SendSmsParams): Promise<string> {
  const { BREVO_API_KEY: apiKey, BREVO_SMS_SENDER } = getEnv()
  if (!apiKey) throw new Error('BREVO_API_KEY não configurado')

  const sender = BREVO_SMS_SENDER ?? 'QUIC'

  const body = JSON.stringify({
    sender,
    recipient: to,
    content: message,
    type: 'transactional',
  })

  const headers = {
    'api-key': apiKey,
    'content-type': 'application/json',
    accept: 'application/json',
  }

  let lastErr: unknown
  for (let attempt = 0; attempt < 3; attempt++) {
    if (attempt > 0) await new Promise(r => setTimeout(r, attempt * 500))
    try {
      const res = await fetch(BREVO_SMS_API, { method: 'POST', headers, body })
      if (!res.ok) {
        const text = await res.text()
        if (res.status < 500) throw new Error(`Brevo SMS ${res.status}: ${text}`)
        lastErr = new Error(`Brevo SMS ${res.status}: ${text}`)
        continue
      }
      const data = await res.json() as { messageId?: string }
      return data.messageId ?? ''
    } catch (err) {
      if (err instanceof Error && err.message.startsWith('Brevo SMS 4')) throw err
      lastErr = err
    }
  }
  throw lastErr
}
```

-- =========================================================
-- Quic Platform — SMS message templates
-- =========================================================
-- Adds default SMS templates for checklist_complete and
-- checklist_start notifications. Plain text, no HTML,
-- <160 chars per message to avoid multi-part SMS.
-- The unique index from 0011 prevents duplicates on re-run.
-- =========================================================

INSERT INTO message_templates
  (organization_id, name, channel, language, template_key, subject, body_template)
SELECT
  '00000000-0000-0000-0000-000000000001',
  'Etapa concluída — SMS',
  'sms',
  'pt',
  'checklist_complete',
  NULL,
  'QUIC: Etapa "{{item_client_label}}" do seu evento concluida ({{progress_percent}}%). Acompanhe em: {{portal_url}}'
WHERE NOT EXISTS (
  SELECT 1 FROM message_templates
  WHERE organization_id = '00000000-0000-0000-0000-000000000001'
    AND channel = 'sms'
    AND language = 'pt'
    AND template_key = 'checklist_complete'
);

INSERT INTO message_templates
  (organization_id, name, channel, language, template_key, subject, body_template)
SELECT
  '00000000-0000-0000-0000-000000000001',
  'Início de preparação de etapa — SMS',
  'sms',
  'pt',
  'checklist_start',
  NULL,
  'QUIC: Iniciamos a etapa "{{item_client_label}}" do seu evento. Acompanhe em: {{portal_url}}'
WHERE NOT EXISTS (
  SELECT 1 FROM message_templates
  WHERE organization_id = '00000000-0000-0000-0000-000000000001'
    AND channel = 'sms'
